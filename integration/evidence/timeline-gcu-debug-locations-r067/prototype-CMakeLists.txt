cmake_minimum_required(VERSION 3.20)
project(FlagPrismDebugScopePrototype LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
add_executable(scope-bridge bridge.cpp)
target_compile_features(scope-bridge PRIVATE cxx_std_17)
target_include_directories(scope-bridge PRIVATE ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
separate_arguments(FLAGS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(scope-bridge PRIVATE ${FLAGS} -fno-rtti)
target_link_libraries(scope-bridge PRIVATE MLIRParser MLIRIR MLIRPass MLIRLLVMIRTransforms MLIRLLVMDialect LLVMSupport)
